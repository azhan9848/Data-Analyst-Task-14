 Clean & standardize datasets (Python)

import pandas as pd
import numpy as np
from dateutil import parser


REGIONS = ['US', 'CA', 'IN']


def load_and_standardize(path):
df = pd.read_csv(path)
# Normalize column names
df.columns = [c.strip().lower() for c in df.columns]


# rename common fields if necessary
rename_map = {}
if 'trending_date' in df.columns:
rename_map['trending_date'] = 'trending_date'
# Example: some datasets use publishedAt
if 'publishedat' in df.columns:
rename_map['publishedat'] = 'publish_time'


df = df.rename(columns=rename_map)


# Standardize dates
def parse_date(x):
try:
return pd.to_datetime(x)
except Exception:
try:
# some files have 17.14.11 format -> convert if needed
return pd.to_datetime(x, format='%y.%d.%m')
except Exception:
return pd.NaT


if 'trending_date' in df.columns:
df['trending_date'] = df['trending_date'].apply(parse_date)
if 'publish_time' in df.columns:
df['publish_time'] = pd.to_datetime(df['publish_time'], errors='coerce')


# Ensure numeric columns
for col in ['views','likes','dislikes','comment_count']:
if col in df.columns:
df[col] = pd.to_numeric(df[col], errors='coerce').fillna(0).astype(int)


return df


# Example: combine
frames = []
for r in REGIONS:
df = load_and_standardize(f'data/{r}_youtube_trending.csv')
df['region'] = r
frames.append(df)


all_df = pd.concat(frames, ignore_index=True)
all_df.to_parquet('data/combined_trending.parquet')

-- 1. Rank categories by average views (overall)
SELECT category_id, COUNT(*) AS videos, AVG(views) AS avg_views
FROM trending
GROUP BY category_id
ORDER BY avg_views DESC;


-- 2. Top categories per region
SELECT region, category_id, AVG(views) AS avg_views
FROM trending
GROUP BY region, category_id
ORDER BY region, avg_views DESC;


-- 3. Trending duration: weeks a video remained trending
-- We first compute first and last trending_date per video
WITH span AS (
SELECT video_id, MIN(trending_date) AS first_seen, MAX(trending_date) AS last_seen
FROM trending
GROUP BY video_id
)
SELECT t.video_id, (julianday(last_seen) - julianday(first_seen)) AS days_trending
FROM span t
ORDER BY days_trending DESC
LIMIT 50;

Sentiment analysis on titles & tags

import pandas as pd
from vaderSentiment.vaderSentiment import SentimentIntensityAnalyzer


analyzer = SentimentIntensityAnalyzer()


df = pd.read_parquet('data/combined_trending.parquet')


def title_sentiment(title):
if pd.isna(title):
return None
s = analyzer.polarity_scores(title)
return s['compound']


# compute
df['title_sentiment'] = df['title'].apply(title_sentiment)
# categorize sentiment
bins = [-1.0, -0.05, 0.05, 1.0]
labels = ['negative','neutral','positive']
df['title_sentiment_label'] = pd.cut(df['title_sentiment'], bins=bins, labels=labels)


# basic tag processing: get top tag words


def extract_tags(tags_field):
if pd.isna(tags_field):
return []
# many datasets have tags like "tag1|tag2" or "\"tag1\""; adapt accordingly
if isinstance(tags_field, str):
return [t.strip().lower() for t in tags_field.replace('"','').split('|') if t.strip()]
return []


from collections import Counter
all_tags = df['tags'].dropna().apply(extract_tags).explode()
Counter(all_tags).most_common(30)


# save
df.to_parquet('data/combined_with_sentiment.parquet')

Exploratory Data Analysis (EDA)

import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns


sns.set(style='whitegrid')
df = pd.read_parquet('data/combined_with_sentiment.parquet')


# 1. Top 10 categories by avg views
cat_avg = df.groupby('category_id').agg({'views':'mean','video_id':'count'}).rename(columns={'video_id':'count'}).sort_values('views', ascending=False).reset_index()
plt.figure(figsize=(10,6))
sns.barplot(data=cat_avg.head(10), x='views', y='category_id')
plt.title('Top 10 categories by average views')
plt.xlabel('Average views')
plt.ylabel('Category')
plt.tight_layout()
plt.show()


# 2. Sentiment distribution across categories
plt.figure(figsize=(12,6))
sns.boxplot(data=df, x='category_id', y='title_sentiment')
plt.xticks(rotation=45)
plt.title('Title sentiment distribution by category')
plt.tight_layout()
plt.show()


# 3. Trending duration histogram
# compute duration
span = df.groupby('video_id').agg(first_seen=('trending_date','min'), last_seen=('trending_date','max')).reset_index()
span['days_trending'] = (span['last_seen'] - span['first_seen']).dt.days
plt.figure(figsize=(8,5))
sns.histplot(span['days_trending'].dropna(), bins=30)
plt.title('Distribution of trending duration (days)')
plt.xlabel('Days on trending')
plt.show()
